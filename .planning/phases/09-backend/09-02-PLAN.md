    1. 创建 framework/agent_framework/viz/agent_runner.py：

       AgentRunner.__init__(self, agent_name: str, bus: EventBus): 存储 _agent_name 和 _bus（per D-08）

       async def wrap(self, loop_gen: AsyncGenerator[LoopEvent, None]) -> AsyncGenerator[LoopEvent, None]:
       - 循环前: await _publish("idle", {})（per D-07 "agent 启动前 → idle"）
       - try 块:
         - async for event in loop_gen:
           - viz = self._map(event)
           - if viz is not None: await self._bus.publish(viz.model_dump())
           - yield event  # 始终透传原始 LoopEvent（per EVNT-07）
       - except Exception:
         - await _publish("error", {"error": str(exc)})
         - raise  # 重新抛出，不吞异常
       - finally:
         - await _publish("shutdown", {})（per D-07 "agent 停止 → shutdown"）

       _map(self, event: LoopEvent) -> VizEvent | None：映射逻辑严格按 D-07：
       - event.type == "step" + data["stop_reason"] == "tool_use" → VizEvent(type="thinking", payload={"step": event.step})
       - event.type == "step" + data["stop_reason"] in ("end_turn", "stop_sequence") → VizEvent(type="done", payload={"step": event.step, **event.data})
       - event.type == "tool_result" → VizEvent(type="tool_call", payload={"step": event.step, **event.data})
       - event.type == "done" → VizEvent(type="done", payload={"step": event.step, **event.data})
       - event.type in ("error", "max_steps") → VizEvent(type="error", payload={"step": event.step, **event.data})
       - 其他 → return None

       _publish(self, viz_type: str, payload: dict) -> None: 辅助方法，构造 VizEvent 并 bus.publish(model_dump())
