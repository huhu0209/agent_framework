    1. 创建 framework/agent_framework/viz/ws_server.py：

       导入: from websockets.asyncio.server import serve, ServerConnection + asyncio + json + logging + time

       模块级 logger = logging.getLogger(__name__)

       async def serve_ws(bus: EventBus, host: str = "localhost", port: int = 8765) -> None:
       - serve() 的 handler_factory 模式：定义内部 _handler(websocket) 闭包捕获 bus
       - async with serve(lambda ws: _handler(ws, bus), host, port) as server:
         - logger.info(f"WebSocket server listening on ws://{host}:{port}")
         - await server.wait_closed()

       async def _handler(websocket: ServerConnection, bus: EventBus) -> None:
       - queue = await bus.subscribe()
       - try:
         - recv_task = asyncio.create_task(_handle_commands(websocket, bus))
         - push_task = asyncio.create_task(_push_events(websocket, queue))
         - done, pending = await asyncio.wait([recv_task, push_task], return_when=asyncio.FIRST_COMPLETED)
         - for task in pending: task.cancel()
         - for task in done: task.result() 处理可能的异常（仅 logging，不吞异常）
       - finally:
         - await bus.unsubscribe(queue)

       async def _push_events(websocket: ServerConnection, queue: asyncio.Queue[dict]) -> None:
       - while True: event = await queue.get(); await websocket.send(json.dumps(event))

       async def _handle_commands(websocket: ServerConnection, bus: EventBus) -> None:
       - async for raw in websocket:
         - try: cmd = json.loads(raw)
         - except json.JSONDecodeError: await _send_response(websocket, False, "Invalid JSON"); continue
         - cmd_type = cmd.get("type")
         - if cmd_type == "start_team": await _handle_start_team(cmd, websocket, bus)
         - elif cmd_type == "stop_team": await _handle_stop_team(cmd, websocket, bus)
         - else: await _send_response(websocket, False, f"Unknown command: {cmd_type}")

       async def _handle_start_team(cmd: dict, websocket: ServerConnection, bus: EventBus) -> None:
       - 从 cmd["agent"] 提取 name/role/system_prompt（per D-12）
       - MVP: 记录到 _active_runners dict（模块级），返回 success=true（per D-09: asyncio.create_task 在后续集成时实现，当前只做命令接收确认）
       - await _send_response(websocket, True)

       async def _handle_stop_team(cmd: dict, websocket: ServerConnection, bus: EventBus) -> None:
       - 从 cmd["name"] 提取 agent 名称
       - MVP: 查找 _active_runners，取消对应 task
       - await _send_response(websocket, True)

       async def _send_response(websocket: ServerConnection, success: bool, error: str | None = None) -> None:
       - response = {"type": "command_response", "success": success}
       - if error: response["error"] = error
       - await websocket.send(json.dumps(response))
